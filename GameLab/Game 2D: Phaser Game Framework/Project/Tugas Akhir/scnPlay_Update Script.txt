// scnPlay TA
// update script

function update() {
    
    // melakukan pengecekan apakah kondisi game
    // sudah dimulai atau belum sama sekali
    if (!this.gameStarted)
    {
        // apabila kondisi game belum dimulai (this.gameStarted bernilai false)
        // maka kode program yang ada dibawahnya tidak akan dijalankan.
        return;
    }
    // mendeteksi apabila tombol arah kanan
    // pada keyboard ditekan
    if (this.cursors.right.isDown) {
        // memberikan nilai percepatan dengan nilai '200' ke karakter utama untuk
        // menggerakkan karakter ke arah kanan dengan bantuan hukum fisika
        this.player.setVelocityX(200);
        // meng-animasikan karakter berlari ke arah kanan
        this.player.anims.play('right', true);
        // modul 11-20
        // mengatur nilai volume menjadi '1' untuk
        // membunyikan suara gerakan karakter 
        this.snd_walk.setVolume(1);
        this.player.setFlipX(false);
    }
    // mendeteksi apabila tombol arah kiri
    // pada keyboard ditekan
    else if (this.cursors.left.isDown) {
        // memberikan nilai percepatan dengan nilai '-200' ke karakter utama untuk
        // menggerakkan karakter ke arah kiri dengan bantuan hukum fisika
        this.player.setVelocityX(-200);
        // meng-animasikan karakter berlari ke arah kiri
        this.player.anims.play('left', true);
        this.player.setFlipX(true);
        // mengatur nilai volume menjadi '1' untuk
        // membunyikan suara gerakan karakter 
        this.snd_walk.setVolume(1);
    }
    // mendeteksi apabila tidak ada tombol arah
    // pada keyboard yang ditekan
    else {
        // memberikan nilai percepatan dengan nilai '0' ke karakter untuk
        // membuat karakter tidak bergerak dengan bantuan hukum fisika
        this.player.setVelocityX(0);
        // meng-animasikan karakter untuk menghadap ke depan
        this.player.anims.play('front');
        // mengatur nilai volume menjadi '0' untuk
        // menghentikan suara gerakan karakter 
        this.snd_walk.setVolume(0);
    }
    // mendeteksi apabila tombol arah bawah pada keyboard
    // ditekan dan kondisi untuk karakter sedang menyentuh pijakan
    if (this.cursors.up.isDown && this.player.body.touching.down) {
        // memberikan nilai percepatan dengan nilai '-620' ke karakter utama untuk
        // menggerakkannya ke arah atas dengan bantuan hukum fisika (lompat)
        this.player.setVelocityY(-650);
        this.snd_jump.play();
    }
}
