Mari belajar cara mengurutkan data dengan mudah dan menyenangkan!

Apa itu Algoritma Pengurutan?

Algoritma pengurutan adalah cara untuk menyusun data dalam urutan tertentu, biasanya dari kecil ke besar (ascending) atau dari besar ke kecil (descending).

💡 Contoh sederhana: Mengurutkan nilai ujian dari yang tertinggi ke terendah, atau menyusun nama siswa berdasarkan abjad.

🎯

Mengapa Penting?

📊 Pencarian Lebih Cepat

Data yang terurut memudahkan pencarian informasi

📈 Analisis Data

Membantu menganalisis pola dan tren data

🎮 Aplikasi Game

Mengurutkan skor pemain atau level

🛒 E-Commerce

Mengurutkan produk berdasarkan harga atau rating

Konsep Dasar

Bubble Sort bekerja seperti gelembung udara yang naik ke permukaan air. Elemen terbesar akan "menggelembung" ke akhir array melalui pertukaran berulang.

🔄 Prinsip: Bandingkan dua elemen bersebelahan, tukar jika tidak sesuai urutan, ulangi sampai semua terurut.

def bubble_sort(arr):
    """
    Fungsi untuk mengurutkan array menggunakan algoritma Bubble Sort
    
    Parameter:
    arr (list): Array yang akan diurutkan
    
    Return:
    list: Array yang sudah terurut
    """
    n = len(arr)
    
    # Loop untuk setiap elemen
    for i in range(n):
        # Flag untuk mengecek apakah ada pertukaran
        swapped = False
        
        # Loop untuk membandingkan elemen bersebelahan
        for j in range(0, n - i - 1):
            # Jika elemen saat ini lebih besar dari elemen berikutnya
            if arr[j] > arr[j + 1]:
                # Tukar posisi
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swapped = True
                
                # Tampilkan proses pertukaran
                print(f"Tukar {arr[j+1]} dan {arr[j]}: {arr}")
        
        # Jika tidak ada pertukaran, array sudah terurut
        if not swapped:
            break
    
    return arr

# Contoh penggunaan
if __name__ == "__main__":
    # Input data dari pengguna
    print("=== PROGRAM BUBBLE SORT ===")
    print("Masukkan angka-angka yang ingin diurutkan (pisahkan dengan spasi)")
    
    try:
        # Input dari keyboard
        input_data = input("Masukkan angka: ")
        numbers = list(map(int, input_data.split()))
        
        print(f"\nData awal: {numbers}")
        print("\nProses pengurutan:")
        
        # Panggil fungsi bubble sort
        sorted_numbers = bubble_sort(numbers.copy())
        
        print(f"\nHasil akhir: {sorted_numbers}")
        
    except ValueError:
        print("Error: Pastikan Anda memasukkan angka yang valid!")

Kompleksitas Waktu

📊 Analisis Performa Bubble Sort:

• Best Case: O(n) - Data sudah terurut

• Average Case: O(n²) - Data acak

• Worst Case: O(n²) - Data urutan terbalik

• Space Complexity: O(1) - Hanya butuh sedikit memori tambahan

💡 Kapan menggunakan Bubble Sort? Cocok untuk data kecil (< 50 elemen) atau untuk pembelajaran konsep dasar.

def selection_sort(arr):
    """
    Fungsi untuk mengurutkan array menggunakan algoritma Selection Sort
    
    Parameter:
    arr (list): Array yang akan diurutkan
    
    Return:
    list: Array yang sudah terurut
    """
    n = len(arr)
    
    # Loop untuk setiap posisi
    for i in range(n):
        # Anggap elemen pertama adalah yang terkecil
        min_index = i
        
        # Cari elemen terkecil di sisa array
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:
                min_index = j
        
        # Tukar elemen terkecil dengan elemen di posisi i
        if min_index != i:
            arr[i], arr[min_index] = arr[min_index], arr[i]
            print(f"Tukar posisi {i} dan {min_index}: {arr}")
    
    return arr

# Program utama

        # Panggil fungsi selection sort

def insertion_sort(arr):
    """
    Fungsi untuk mengurutkan array menggunakan algoritma Insertion Sort
    
    Parameter:
    arr (list): Array yang akan diurutkan
    
    Return:
    list: Array yang sudah terurut
    """
    # Mulai dari elemen kedua (indeks 1)
    for i in range(1, len(arr)):
        # Elemen yang akan disisipkan
        key = arr[i]
        
        # Posisi untuk membandingkan dengan elemen sebelumnya
        j = i - 1
        
        # Geser elemen yang lebih besar dari key ke kanan
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        
        # Sisipkan key di posisi yang tepat
        arr[j + 1] = key
        
        # Tampilkan proses penyisipan
        print(f"Sisipkan {key} di posisi {j + 1}: {arr}")
    
    return arr

import time
import random

class SortingApp:
    """
    Aplikasi lengkap untuk belajar algoritma pengurutan
    Menggabungkan Bubble Sort, Selection Sort, dan Insertion Sort
    """
    
    def __init__(self):
        self.data = []
    
    def bubble_sort(self, arr, show_steps=True):
        """Implementasi Bubble Sort dengan opsi menampilkan langkah"""
        arr = arr.copy()
        n = len(arr)
        steps = []
        
        for i in range(n):
            swapped = False
            for j in range(0, n - i - 1):
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
                    swapped = True
                    if show_steps:
                        steps.append(f"Tukar {arr[j]} dan {arr[j+1]}: {arr.copy()}")
            
            if not swapped:
                break
        
        return arr, steps
    
    def selection_sort(self, arr, show_steps=True):
        """Implementasi Selection Sort dengan opsi menampilkan langkah"""
        arr = arr.copy()
        n = len(arr)
        steps = []
        
        for i in range(n):
            min_index = i
            for j in range(i + 1, n):
                if arr[j] < arr[min_index]:
                    min_index = j
            
            if min_index != i:
                arr[i], arr[min_index] = arr[min_index], arr[i]
                if show_steps:
                    steps.append(f"Pilih minimum {arr[i]}, tukar posisi {i} dan {min_index}: {arr.copy()}")
        
        return arr, steps
    
    def insertion_sort(self, arr, show_steps=True):
        """Implementasi Insertion Sort dengan opsi menampilkan langkah"""
        arr = arr.copy()
        steps = []
        
        for i in range(1, len(arr)):
            key = arr[i]
            j = i - 1
            
            while j >= 0 and arr[j] > key:
                arr[j + 1] = arr[j]
                j -= 1
            
            arr[j + 1] = key
            if show_steps:
                steps.append(f"Sisipkan {key} di posisi {j + 1}: {arr.copy()}")
        
        return arr, steps
    
    def measure_time(self, sort_function, data):
        """Mengukur waktu eksekusi algoritma"""
        start_time = time.time()
        result, _ = sort_function(data, show_steps=False)
        end_time = time.time()
        return result, (end_time - start_time) * 1000  # dalam milidetik

    def display_menu(self):
        """Menampilkan menu utama aplikasi"""
        print("\n" + "="*60)
        print("🎯 APLIKASI PEMBELAJARAN ALGORITMA SORTING")
        print("="*60)
        print("1. 🫧 Bubble Sort")
        print("2. 🎯 Selection Sort")
        print("3. 📝 Insertion Sort")
        print("4. ⚡ Bandingkan Semua Algoritma")
        print("5. 📊 Lihat Data Saat Ini")
        print("6. 🔄 Input Data Baru")
        print("7. 🎲 Generate Data Random")
        print("0. ❌ Keluar")
        print("="*60)
    
    def input_data(self):
        """Input data dari pengguna"""
        try:
            input_str = input("Masukkan angka (pisahkan dengan spasi): ")
            self.data = list(map(int, input_str.split()))
            print(f"✅ Data berhasil disimpan: {self.data}")
        except ValueError:
            print("❌ Error: Pastikan input berupa angka yang valid!")
    
    def generate_random_data(self):
        """Generate data random"""
        try:
            size = int(input("Jumlah elemen (5-20): "))
            if 5 <= size <= 20:
                self.data = [random.randint(1, 100) for _ in range(size)]
                print(f"🎲 Data random: {self.data}")
            else:
                print("❌ Jumlah elemen harus antara 5-20")
        except ValueError:
            print("❌ Input tidak valid!")
    
    def run(self):
        """Menjalankan aplikasi utama"""
        print("🎉 Selamat datang di Aplikasi Pembelajaran Sorting!")
        
        while True:
            self.display_menu()
            
            try:
                choice = input("Pilih menu (0-7): ")
                
                if choice == '0':
                    print("👋 Terima kasih telah belajar! Sampai jumpa!")
                    break
                
                elif choice == '6':
                    self.input_data()
                
                elif choice == '7':
                    self.generate_random_data()
                
                elif choice == '5':
                    if self.data:
                        print(f"📊 Data saat ini: {self.data}")
                    else:
                        print("❌ Belum ada data. Silakan input data terlebih dahulu.")
                
                elif choice in ['1', '2', '3', '4']:
                    if not self.data:
                        print("❌ Belum ada data. Silakan input data terlebih dahulu.")
                        continue
                    
                    if choice == '1':
                        print(f"\n🫧 BUBBLE SORT - Data: {self.data}")
                        result, steps = self.bubble_sort(self.data)
                        for step in steps:
                            print(step)
                        print(f"✅ Hasil: {result}")
                    
                    elif choice == '2':
                        print(f"\n🎯 SELECTION SORT - Data: {self.data}")
                        result, steps = self.selection_sort(self.data)
                        for step in steps:
                            print(step)
                        print(f"✅ Hasil: {result}")
                    
                    elif choice == '3':
                        print(f"\n📝 INSERTION SORT - Data: {self.data}")
                        result, steps = self.insertion_sort(self.data)
                        for step in steps:
                            print(step)
                        print(f"✅ Hasil: {result}")
                    
                    elif choice == '4':
                        print(f"\n⚡ PERBANDINGAN ALGORITMA - Data: {self.data}")
                        algorithms = [
                            ("Bubble Sort", self.bubble_sort),
                            ("Selection Sort", self.selection_sort),
                            ("Insertion Sort", self.insertion_sort)
                        ]
                        
                        print("-" * 50)
                        for name, func in algorithms:
                            result, time_ms = self.measure_time(func, self.data)
                            print(f"{name:15} | {time_ms:8.4f} ms | {result}")
                        print("-" * 50)
                
                else:
                    print("❌ Pilihan tidak valid!")
                    
            except KeyboardInterrupt:
                print("\n\n👋 Program dihentikan. Sampai jumpa!")
                break
            except Exception as e:
                print(f"❌ Terjadi error: {e}")

# Menjalankan aplikasi
if __name__ == "__main__":
    app = SortingApp()
    app.run()
