Pengenalan GUI Java Swing

🔍 Apa itu Java Swing?

Java Swing adalah toolkit GUI (Graphical User Interface) yang merupakan bagian dari Java Foundation Classes (JFC). Swing menyediakan komponen-komponen untuk membuat aplikasi desktop yang kaya fitur dan interaktif.

└── javax.swing.JComponent
├── JButton

├── JLabel
├── JCheckBox
├── JRadioButton
└── JComboBox

import javax.swing.*;
import java.awt.event.*;

public class MainFrame extends JFrame {
    
    // Deklarasi komponen
    private JButton btnSubmit;
    private JTextField txtNama;
    private JLabel lblHasil;
    
    public MainFrame() {
        initComponents(); // NetBeans auto-generated
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setTitle("Aplikasi GUI Java");
    }
    
    // Event handler - NetBeans auto-generated method
    private void btnSubmitActionPerformed(ActionEvent evt) {
        String nama = txtNama.getText().trim();
        if (!nama.isEmpty()) {
            lblHasil.setText("Halo, " + nama + "!");
        } else {
            JOptionPane.showMessageDialog(this, "Nama tidak boleh kosong!");
        }
    }
}

}

import javax.swing.JOptionPane;

// Method untuk validasi dan proses input
private void prosesInput() {
    // Mengambil input dari user
    String nama = txtNama.getText().trim();
    
    // Validasi input kosong
    if (nama.isEmpty()) {
        JOptionPane.showMessageDialog(this, 
            "Nama tidak boleh kosong!", 
            "Peringatan", 
            JOptionPane.WARNING_MESSAGE);
        txtNama.requestFocus();
        return;
    }
    
    // Validasi panjang minimal
    if (nama.length() < 3) {
        JOptionPane.showMessageDialog(this, 
            "Nama minimal 3 karakter!", 
            "Peringatan", 
            JOptionPane.WARNING_MESSAGE);
        txtNama.selectAll(); // Select semua teks
        return;
    }
    
    // Validasi karakter khusus (hanya huruf dan spasi)
    if (!nama.matches("[a-zA-Z\\s]+")) {
        JOptionPane.showMessageDialog(this, 
            "Nama hanya boleh berisi huruf dan spasi!");
        return;
    }
    
    // Proses berhasil - tampilkan hasil
    lblHasil.setText("Selamat datang, " + nama + "!");
    txtNama.setEditable(false); // Lock input
}

import java.util.ArrayList;
import java.util.List;
import javax.swing.JOptionPane;

// Membuat checkbox dengan initial state
JCheckBox chkSetuju = new JCheckBox("Saya setuju dengan syarat dan ketentuan");
JCheckBox chkMembaca = new JCheckBox("Membaca", true); // Default selected
JCheckBox chkOlahraga = new JCheckBox("Olahraga");
JCheckBox chkMusik = new JCheckBox("Musik");

// Method untuk validasi dan mengumpulkan pilihan
private void prosesHobi() {
    List<String> selectedHobbies = new ArrayList<>();
    
    if (chkMembaca.isSelected()) selectedHobbies.add("Membaca");
    if (chkOlahraga.isSelected()) selectedHobbies.add("Olahraga");
    if (chkMusik.isSelected()) selectedHobbies.add("Musik");
    
    // Validasi minimal satu hobi dipilih
    if (selectedHobbies.isEmpty()) {
        JOptionPane.showMessageDialog(this, "Pilih minimal satu hobi!");
        return;
    }
    
    // Tampilkan hasil
    String hasil = "Hobi yang dipilih: " + String.join(", ", selectedHobbies);
    lblHasil.setText(hasil);
}

// Event handling untuk enable/disable button
chkSetuju.addActionListener(e -> {
    btnDaftar.setEnabled(chkSetuju.isSelected());
    if (chkSetuju.isSelected()) {
        btnDaftar.setText("Daftar Sekarang");
    } else {
        btnDaftar.setText("Setujui Dulu");
    }
});

import javax.swing.ButtonGroup;
import javax.swing.JOptionPane;

// Membuat ButtonGroup untuk jenis kelamin
ButtonGroup groupGender = new ButtonGroup();

// Membuat RadioButton
JRadioButton rbPria = new JRadioButton("Laki-laki");
JRadioButton rbWanita = new JRadioButton("Perempuan");

// PENTING: Menambahkan ke group
groupGender.add(rbPria);
groupGender.add(rbWanita);

// Set default selection
rbPria.setSelected(true);

// Method untuk mendapatkan pilihan
private String getSelectedGender() {
    if (rbPria.isSelected()) {
        return "Laki-laki";
    } else if (rbWanita.isSelected()) {
        return "Perempuan";
    } else {
        return null; // Tidak ada yang dipilih
    }
}

// Event handling untuk perubahan pilihan
rbPria.addActionListener(e -> {
    lblGender.setText("Jenis Kelamin: Laki-laki");
});

rbWanita.addActionListener(e -> {
    lblGender.setText("Jenis Kelamin: Perempuan");
});

// Method untuk validasi dan mendapatkan pilihan
private boolean validateComboBox() {
    int selectedIndex = cbBulan.getSelectedIndex();
    if (selectedIndex == 0) {
        JOptionPane.showMessageDialog(this, 
            "Silakan pilih bulan!", 
            "Peringatan", 
            JOptionPane.WARNING_MESSAGE);
        cbBulan.requestFocus();
        return false;
    }
    return true;
}

// Event handling dengan validasi
cbBulan.addActionListener(e -> {
    String pilihan = (String) cbBulan.getSelectedItem();
    if (cbBulan.getSelectedIndex() > 0) {
        lblHasil.setText("Bulan dipilih: " + pilihan);
    }
});

💡 Tips & Best Practices

✅ Do's (Yang Harus Dilakukan)

  • • Selalu validasi input sebelum diproses
  • • Gunakan ButtonGroup untuk RadioButton
  • • Berikan feedback yang jelas kepada user
  • • Gunakan placeholder text yang informatif
  • • Set default values yang masuk akal
  • • Disable komponen saat tidak diperlukan

❌ Don'ts (Yang Harus Dihindari)

  • • Jangan lupa trim() untuk input teks
  • • Jangan biarkan ComboBox tanpa validasi
  • • Jangan gunakan System.out.println() di GUI
  • • Jangan lupa handle null values
  • • Jangan buat UI yang membingungkan
  • • Jangan abaikan accessibility

Tugas Praktik

Saatnya menerapkan ilmu yang sudah dipelajari!

🎯 Tugas 1: Form Pendaftaran Siswa

📝 Instruksi:

Buatlah form pendaftaran siswa dengan komponen berikut:

  • • JTextField untuk Nama Lengkap
  • • JTextField untuk NIS (Nomor Induk Siswa)
  • • JRadioButton untuk Jenis Kelamin (Laki-laki/Perempuan)
  • • JComboBox untuk Kelas (X, XI, XII)
  • • JCheckBox untuk Ekstrakurikuler (Pramuka, Basket, Musik, dll)
  • • JButton untuk Submit dan Reset
  • • JLabel untuk menampilkan hasil

✅ Kriteria Penilaian:

  • • Semua komponen berfungsi dengan baik
  • • Validasi input (nama tidak boleh kosong, NIS harus angka)
  • • RadioButton menggunakan ButtonGroup
  • • Tombol Reset menghapus semua input
  • • Layout rapi dan user-friendly

🎯 Tugas 2: Kalkulator Sederhana

🧮 Instruksi:

Buatlah kalkulator sederhana dengan fitur:

  • • JTextField untuk Angka 1

  • • JLabel untuk menampilkan hasil

✅ Fitur Tambahan:

  • • Validasi input harus berupa angka
  • • Penanganan pembagian dengan nol
