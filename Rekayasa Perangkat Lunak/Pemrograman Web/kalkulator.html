Langkah 3: JavaScript untuk Interaksi

Mari buat kalkulator kita hidup dengan JavaScript!

🚀 Step 1: Fungsi Dasar JavaScript

Tambahkan script di bagian bawah HTML atau buat file script.js:

📄 script.js - Basic Functions JavaScript 
let display = document.getElementById('result');
let currentInput = '';
let shouldResetDisplay = false;

// Fungsi untuk menambahkan input ke display
function appendToDisplay(value) {
    if (shouldResetDisplay) {
        currentInput = '';
        shouldResetDisplay = false;
    }
    
    currentInput += value;
    display.value = currentInput;
}

// Fungsi untuk membersihkan semua
function clearAll() {
    currentInput = '';
    display.value = '0';
    shouldResetDisplay = false;
}

// Fungsi untuk menghapus karakter terakhir
function deleteLast() {
    currentInput = currentInput.slice(0, -1);
    display.value = currentInput || '0';
}

🧮 Step 2: Fungsi Perhitungan Utama

Buat fungsi untuk menghitung hasil dengan dukungan fungsi scientific:

🔢 Calculate Function JavaScript 
function calculate() {
    try {
        let expression = currentInput;
        
        // Ganti fungsi scientific dengan Math functions
        expression = expression.replace(/sin\(/g, 'Math.sin(');
        expression = expression.replace(/cos\(/g, 'Math.cos(');
        expression = expression.replace(/tan\(/g, 'Math.tan(');
        expression = expression.replace(/log\(/g, 'Math.log10(');
        expression = expression.replace(/ln\(/g, 'Math.log(');
        expression = expression.replace(/sqrt\(/g, 'Math.sqrt(');
        
        // Handle pangkat
        expression = expression.replace(/\^2/g, '**2');
        expression = expression.replace(/\^/g, '**');
        
        // Evaluasi expression
        let result = eval(expression);
        
        // Format hasil
        if (isNaN(result) || !isFinite(result)) {
            display.value = 'Error';
        } else {
            display.value = parseFloat(result.toFixed(10));
        }
        
        currentInput = display.value.toString();
        shouldResetDisplay = true;
        
    } catch (error) {
        display.value = 'Error';
        currentInput = '';
        shouldResetDisplay = true;
    }
}

⌨️ Step 4: Keyboard Support

Tambahkan dukungan keyboard agar lebih mudah digunakan:

⌨️ Keyboard Events JavaScript 
// Event listener untuk keyboard
document.addEventListener('keydown', function(event) {
    const key = event.key;
    
    // Angka dan operasi dasar
    if ('0123456789+-*/.'.includes(key)) {
        appendToDisplay(key);
    }
    
    // Enter untuk calculate
    else if (key === 'Enter') {
        event.preventDefault();
        calculate();
    }
    
    // Escape untuk clear
    else if (key === 'Escape') {
        clearAll();
    }
    
    // Backspace untuk delete
    else if (key === 'Backspace') {
        deleteLast();
    }
});

// Mencegah input langsung ke display
display.addEventListener('keydown', function(event) {
    event.preventDefault();
});

Langkah 4: PHP Backend untuk Perhitungan Server-Side

Tambahkan kekuatan server-side processing dengan PHP!

🚀 Step 1: Setup PHP Calculator Class

Buat file calculator.php dengan class untuk perhitungan:

📄 calculator.php - Class Setup PHP 
<?php
class ScientificCalculator {
    
    // Fungsi untuk operasi dasar
    public function add($a, $b) {
        return $a + $b;
    }
    
    public function subtract($a, $b) {
        return $a - $b;
    }
    
    public function multiply($a, $b) {
        return $a * $b;
    }
    
    public function divide($a, $b) {
        if ($b == 0) {
            throw new Exception("Tidak bisa dibagi dengan nol!");
        }
        return $a / $b;
    }
    
    // Fungsi pangkat
    public function power($base, $exponent) {
        return pow($base, $exponent);
    }
    
    // Fungsi akar kuadrat
    public function squareRoot($number) {
        if ($number < 0) {
            throw new Exception("Tidak bisa akar kuadrat dari bilangan negatif!");
        }
        return sqrt($number);
    }
}
?>

📐 Step 2: Fungsi Trigonometri

Tambahkan fungsi trigonometri dalam class PHP:

📐 Trigonometry Functions PHP 
// Tambahkan dalam class ScientificCalculator

// Konversi derajat ke radian
private function toRadians($degrees) {
    return $degrees * (M_PI / 180);
}

// Fungsi Sinus
public function sine($angle) {
    return sin($this->toRadians($angle));
}

// Fungsi Cosinus
public function cosine($angle) {
    return cos($this->toRadians($angle));
}

// Fungsi Tangen
public function tangent($angle) {
    $radians = $this->toRadians($angle);
    if (cos($radians) == 0) {
        throw new Exception("Tangen tidak terdefinisi untuk sudut ini!");
    }
    return tan($radians);
}

// Fungsi Logaritma
public function logarithm($number, $base = 10) {
    if ($number <= 0) {
        throw new Exception("Logaritma hanya untuk bilangan positif!");
    }
    if ($base == 10) {
        return log10($number);
    } else {
        return log($number) / log($base);
    }
}

// Fungsi Logaritma Natural
public function naturalLog($number) {
    if ($number <= 0) {
        throw new Exception("Ln hanya untuk bilangan positif!");
    }
    return log($number);
}

🌐 Step 3: API Endpoint untuk AJAX

Buat file api.php untuk menerima request dari JavaScript:

📄 api.php - AJAX Handler PHP 
<?php
header('Content-Type: application/json');
require_once 'calculator.php';

$calculator = new ScientificCalculator();
$response = array();

try {
    if ($_POST['action']) {
        $action = $_POST['action'];
        
        switch ($action) {
            case 'calculate':
                $expression = $_POST['expression'];
                $result = $calculator->evaluateExpression($expression);
                $response['success'] = true;
                $response['result'] = $result;
                break;
                
            case 'sin':
                $angle = floatval($_POST['value']);
                $result = $calculator->sine($angle);
                $response['success'] = true;
                $response['result'] = $result;
                break;
                
            case 'cos':
                $angle = floatval($_POST['value']);
                $result = $calculator->cosine($angle);
                $response['success'] = true;
                $response['result'] = $result;
                break;
                
            case 'tan':
                $angle = floatval($_POST['value']);
                $result = $calculator->tangent($angle);
                $response['success'] = true;
                $response['result'] = $result;
                break;
                
            case 'log':
                $number = floatval($_POST['value']);
                $result = $calculator->logarithm($number);
                $response['success'] = true;
                $response['result'] = $result;
                break;
                
            case 'sqrt':
                $number = floatval($_POST['value']);
                $result = $calculator->squareRoot($number);
                $response['success'] = true;
                $response['result'] = $result;
                break;
                
            default:
                $response['success'] = false;
                $response['error'] = 'Aksi tidak dikenal';
        }
    }
} catch (Exception $e) {
    $response['success'] = false;
    $response['error'] = $e->getMessage();
}

echo json_encode($response);
?>

🔄 Step 4: Integrasi AJAX dengan JavaScript

Update JavaScript untuk menggunakan PHP backend:

🔄 AJAX Integration JavaScript 
// Fungsi untuk mengirim request ke PHP
function calculateWithPHP(action, value = null) {
    const formData = new FormData();
    formData.append('action', action);
    
    if (value !== null) {
        formData.append('value', value);
    } else {
        formData.append('expression', currentInput);
    }
    
    fetch('api.php', {
        method: 'POST',
        body: formData
    })
    .then(response => response.json())
    .then(data => {
        if (data.success) {
            display.value = parseFloat(data.result.toFixed(10));
            currentInput = display.value.toString();
            shouldResetDisplay = true;
        } else {
            display.value = 'Error: ' + data.error;
            currentInput = '';
            shouldResetDisplay = true;
        }
    })
    .catch(error => {
        display.value = 'Network Error';
        console.error('Error:', error);
    });
}

// Update fungsi scientific untuk menggunakan PHP
function calculateSinPHP() {
    let value = parseFloat(currentInput);
    calculateWithPHP('sin', value);
}

function calculateCosPHP() {
    let value = parseFloat(currentInput);
    calculateWithPHP('cos', value);
}

function calculateTanPHP() {
    let value = parseFloat(currentInput);
    calculateWithPHP('tan', value);
}
