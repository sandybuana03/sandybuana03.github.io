Membuat Database & Tabel

Tempat menyimpan data aplikasi kita!

1

Buka phpMyAdmin

Jalankan XAMPP, start Apache & MySQL, lalu buka browser dan ketik: localhost/phpmyadmin

2

Buat Database Baru

Klik "New" di sidebar kiri, lalu buat database dengan nama crud_siswa

📝 SQL Query:
CREATE DATABASE crud_siswa;
3

Buat Tabel Siswa

Pilih database crud_siswa, lalu buat tabel dengan struktur berikut:

📋 SQL Query untuk Tabel:
CREATE TABLE siswa (
  id INT(11) AUTO_INCREMENT PRIMARY KEY,
  nama VARCHAR(100) NOT NULL,
  kelas VARCHAR(10) NOT NULL,
  jurusan VARCHAR(50) NOT NULL,
  alamat TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Membuat Model

Model adalah jembatan antara aplikasi dan database!

1

Buat File Model

Buat file baru di folder application/models/ dengan nama Siswa_model.php

📋 application/models/Siswa_model.php
<?php
defined('BASEPATH') OR exit('No direct script access allowed');

class Siswa_model extends CI_Model {

  public function __construct() {
    parent::__construct();
    $this->load->database();
  }

  // Fungsi untuk mengambil semua data siswa
  public function get_all_siswa() {
    $query = $this->db->get('siswa');
    return $query->result();
  }

  // Fungsi untuk mengambil data siswa berdasarkan ID
  public function get_siswa_by_id($id) {
    $query = $this->db->get_where('siswa', array('id' => $id));
    return $query->row();
  }

  // Fungsi untuk menambah data siswa baru
  public function insert_siswa($data) {
    return $this->db->insert('siswa', $data);
  }

  // Fungsi untuk mengupdate data siswa
  public function update_siswa($id, $data) {
    $this->db->where('id', $id);
    return $this->db->update('siswa', $data);
  }

  // Fungsi untuk menghapus data siswa
  public function delete_siswa($id) {
    $this->db->where('id', $id);
    return $this->db->delete('siswa');
  }
}

💡 Penjelasan Fungsi Model

get_all_siswa()
Mengambil semua data siswa dari database untuk ditampilkan di halaman utama.
get_siswa_by_id()
Mengambil data siswa tertentu berdasarkan ID untuk edit atau detail.
insert_siswa()
Menambahkan data siswa baru ke database.
update_siswa()
Mengubah data siswa yang sudah ada di database.

Membuat Controller

Controller mengatur logika aplikasi dan menghubungkan Model dengan View!

1

Buat File Controller

Buat file baru di folder application/controllers/ dengan nama Siswa.php

📋 application/controllers/Siswa.php
<?php
defined('BASEPATH') OR exit('No direct script access allowed');

class Siswa extends CI_Controller {

  public function __construct() {
    parent::__construct();
    $this->load->model('Siswa_model');
    $this->load->helper('url');
  }

  // Halaman utama - menampilkan daftar siswa
  public function index() {
    $data['siswa'] = $this->Siswa_model->get_all_siswa();
    $this->load->view('siswa/index', $data);
  }

  // Halaman tambah siswa
  public function create() {
    $this->load->view('siswa/create');
  }

  // Proses simpan data siswa baru
  public function store() {
    $data = array(
      'nama' => $this->input->post('nama'),
      'kelas' => $this->input->post('kelas'),
      'jurusan' => $this->input->post('jurusan'),
      'alamat' => $this->input->post('alamat')
    );

    if($this->Siswa_model->insert_siswa($data)) {
      redirect('siswa');
    }
  }

  // Halaman edit siswa
  public function edit($id) {
    $data['siswa'] = $this->Siswa_model->get_siswa_by_id($id);
    $this->load->view('siswa/edit', $data);
  }

  // Proses update data siswa
  public function update($id) {
    $data = array(
      'nama' => $this->input->post('nama'),
      'kelas' => $this->input->post('kelas'),
      'jurusan' => $this->input->post('jurusan'),
      'alamat' => $this->input->post('alamat')
    );

    if($this->Siswa_model->update_siswa($id, $data)) {
      redirect('siswa');
    }
  }

  // Proses hapus data siswa
  public function delete($id) {
    if($this->Siswa_model->delete_siswa($id)) {
      redirect('siswa');
    }
  }
}

🎯 Penjelasan Method Controller

R
index() - READ
Menampilkan halaman utama dengan daftar semua siswa
C
create() & store() - CREATE
Menampilkan form tambah siswa dan memproses penyimpanan data
U
edit() & update() - UPDATE
Menampilkan form edit siswa dan memproses perubahan data
D
delete() - DELETE
Menghapus data siswa berdasarkan ID

Troubleshooting & Tips

❌ Error: 404 Page Not Found
• Pastikan file .htaccess sudah dibuat
• Cek konfigurasi base_url di config.php
• Pastikan mod_rewrite Apache aktif
⚠️ Error: Database Connection
• Pastikan MySQL sudah running di XAMPP
• Cek nama database di config/database.php
• Pastikan autoload database sudah diset
💡 Tips Debugging
• Aktifkan error reporting di index.php
• Gunakan var_dump() untuk debug data
• Cek log error di application/logs/
🔒 Tips Keamanan & Validasi
• Gunakan $this->input->post() untuk input
• Tambahkan validasi form di controller
• Escape output dengan htmlspecialchars()
• Gunakan CSRF protection untuk form
