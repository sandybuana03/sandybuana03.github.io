# Kalkulator Sederhana - Versi Lengkap dengan Validasi
import math  # Import library untuk fungsi matematika lanjutan

def tampilkan_menu():
    """Menampilkan menu pilihan operasi"""
    print("\n" + "="*40)
    print("🧮 KALKULATOR PINTAR PYTHON 🧮")
    print("="*40)
    print("Pilih operasi yang diinginkan:")
    print("1. ➕ Penjumlahan")
    print("2. ➖ Pengurangan") 
    print("3. ✖️  Perkalian")
    print("4. ➗ Pembagian")
    print("5. 🔢 Pangkat")
    print("6. 📊 Modulus (Sisa Bagi)")
    print("7. √  Akar Kuadrat")
    print("8. 🚪 Keluar")
    print("="*40)

def validasi_angka(prompt):
    """Fungsi untuk validasi input angka"""
    while True:
        try:
            angka = float(input(prompt))
            return angka
        except ValueError:
            print("❌ Input tidak valid! Masukkan angka yang benar.")

def kalkulator_pintar():
    """Fungsi utama kalkulator dengan berbagai fitur"""
    print("🎉 Selamat datang di Kalkulator Pintar!")
    
    while True:
        tampilkan_menu()
        
        try:
            pilihan = input("Masukkan pilihan (1-8): ").strip()
            
            if pilihan == '8':
                print("👋 Terima kasih telah menggunakan Kalkulator Pintar!")
                break
            
            if pilihan not in ['1', '2', '3', '4', '5', '6', '7']:
                print("❌ Pilihan tidak valid! Pilih angka 1-8.")
                continue
            
            # Input angka berdasarkan operasi
            if pilihan == '7':  # Akar kuadrat hanya butuh 1 angka
                angka1 = validasi_angka("Masukkan angka: ")
                if angka1 < 0:
                    print("❌ Tidak bisa menghitung akar kuadrat dari angka negatif!")
                    continue
            else:  # Operasi lain butuh 2 angka
                angka1 = validasi_angka("Masukkan angka pertama: ")
                angka2 = validasi_angka("Masukkan angka kedua: ")
            
            # Proses perhitungan
            if pilihan == '1':
                hasil = angka1 + angka2
                print(f"✅ Hasil: {angka1} + {angka2} = {hasil}")
                
            elif pilihan == '2':
                hasil = angka1 - angka2
                print(f"✅ Hasil: {angka1} - {angka2} = {hasil}")
                
            elif pilihan == '3':
                hasil = angka1 * angka2
                print(f"✅ Hasil: {angka1} × {angka2} = {hasil}")
                
            elif pilihan == '4':
                if angka2 == 0:
                    print("❌ Error: Tidak bisa dibagi dengan nol!")
                else:
                    hasil = angka1 / angka2
                    print(f"✅ Hasil: {angka1} ÷ {angka2} = {hasil}")
                    
            elif pilihan == '5':
                hasil = angka1 ** angka2
                print(f"✅ Hasil: {angka1} ^ {angka2} = {hasil}")
                
            elif pilihan == '6':
                if angka2 == 0:
                    print("❌ Error: Tidak bisa modulus dengan nol!")
                else:
                    hasil = angka1 % angka2
                    print(f"✅ Hasil: {angka1} mod {angka2} = {hasil}")
                    
            elif pilihan == '7':
                hasil = math.sqrt(angka1)
                print(f"✅ Hasil: √{angka1} = {hasil}")
            
            # Tanya apakah ingin melanjutkan
            lanjut = input("\n🔄 Ingin melakukan perhitungan lagi? (y/n): ").lower()
            if lanjut not in ['y', 'yes', 'ya']:
                print("👋 Terima kasih telah menggunakan Kalkulator Pintar!")
                break
                
        except KeyboardInterrupt:
            print("\n\n👋 Program dihentikan oleh user. Sampai jumpa!")
            break
        except Exception as e:
            print(f"❌ Terjadi error: {e}")

# Jalankan program utama
if __name__ == "__main__":
    kalkulator_pintar()

# Sistem Manajemen Data Siswa
class ManajemenSiswa:
    def __init__(self):
        self.daftar_siswa = []  # List untuk menyimpan data siswa
    
    def tambah_siswa(self, nama):
        """Menambah siswa baru ke dalam list"""
        if nama.strip():  # Cek apakah nama tidak kosong
            self.daftar_siswa.append(nama.strip())
            print(f"✅ Siswa '{nama}' berhasil ditambahkan!")
        else:
            print("❌ Nama tidak boleh kosong!")
    
    def lihat_semua_siswa(self):
        """Menampilkan semua siswa"""
        if self.daftar_siswa:
            print("\n📋 DAFTAR SISWA:")
            for i, nama in enumerate(self.daftar_siswa, 1):
                print(f"{i}. {nama}")
        else:
            print("📭 Belum ada data siswa")
    
    def cari_siswa(self, nama_cari):
        """Mencari siswa berdasarkan nama"""
        hasil = []
        for nama in self.daftar_siswa:
            if nama_cari.lower() in nama.lower():
                hasil.append(nama)
        
        if hasil:
            print(f"\n🔍 Ditemukan {len(hasil)} siswa:")
            for nama in hasil:
                print(f"- {nama}")
        else:
            print(f"❌ Siswa dengan nama '{nama_cari}' tidak ditemukan")
    
    def hapus_siswa(self, nama):
        """Menghapus siswa dari list"""
        if nama in self.daftar_siswa:
            self.daftar_siswa.remove(nama)
            print(f"✅ Siswa '{nama}' berhasil dihapus!")
        else:
            print(f"❌ Siswa '{nama}' tidak ditemukan!")
    
    def jumlah_siswa(self):
        """Mengembalikan jumlah siswa"""
        return len(self.daftar_siswa)

# Contoh penggunaan
def main():
    sistem = ManajemenSiswa()
    
    while True:
        print("\n=== SISTEM MANAJEMEN SISWA ===")
        print("1. Tambah Siswa")
        print("2. Lihat Semua Siswa") 
        print("3. Cari Siswa")
        print("4. Hapus Siswa")
        print("5. Keluar")
        
        pilihan = input("Pilih menu (1-5): ")
        
        if pilihan == '1':
            nama = input("Masukkan nama siswa: ")
            sistem.tambah_siswa(nama)
        elif pilihan == '2':
            sistem.lihat_semua_siswa()
        elif pilihan == '3':
            nama_cari = input("Masukkan nama yang dicari: ")
            sistem.cari_siswa(nama_cari)
        elif pilihan == '4':
            nama = input("Masukkan nama siswa yang akan dihapus: ")
            sistem.hapus_siswa(nama)
        elif pilihan == '5':
            print("👋 Terima kasih!")
            break
        else:
            print("❌ Pilihan tidak valid!")

if __name__ == "__main__":
    main()
