Pengenalan Konsep 📖

Mari kenali dulu apa itu Abstract Class dan Interface

Apa itu Abstract Class? 🎭

Abstract Class adalah class yang tidak bisa dibuat objeknya secara langsung. Ia seperti "template" atau "cetakan" yang harus diwarisi oleh class lain.

Analogi: 🏠

Abstract Class seperti blueprint rumah. Kamu tidak bisa tinggal di blueprint, tapi kamu bisa membangun rumah berdasarkan blueprint tersebut!

Apa itu Interface? 🔌

Interface adalah "kontrak" yang menentukan method apa saja yang harus ada di dalam class yang mengimplementasikannya.

Analogi: 🎮

Interface seperti controller game. Semua controller harus punya tombol A, B, X, Y, tapi cara kerjanya bisa berbeda-beda tergantung mereknya!

Perbedaan Utama 🤔

Abstract Class

  • • Bisa punya method yang sudah diimplementasi
  • • Bisa punya atribut/variabel
  • • Diwarisi dengan inheritance
  • • Satu class hanya bisa inherit satu abstract class

Interface

  • • Hanya berisi method yang belum diimplementasi
  • • Tidak bisa punya atribut
  • • Diimplementasi dengan implements
  • • Satu class bisa implement banyak interface

Abstract Class 🎭

Pelajari cara membuat dan menggunakan Abstract Class

Cara Membuat Abstract Class 📝

Di Python, kita menggunakan modul abc (Abstract Base Classes) untuk membuat abstract class:

Python hewan.py 
from abc import ABC, abstractmethod

class Hewan(ABC):  # Abstract Class
    def __init__(self, nama):
        self.nama = nama
    
    # Method biasa (sudah diimplementasi)
    def info(self):
        return f"Ini adalah {self.nama}"
    
    # Abstract method (harus diimplementasi di child class)
    @abstractmethod
    def bersuara(self):
        pass
    
    @abstractmethod
    def bergerak(self):
        pass

class Kucing(Hewan):

Interface 🔌

Memahami konsep Interface dalam Python

Interface di Python 🐍

Python tidak memiliki keyword interface seperti Java, tapi kita bisa membuat interface menggunakan abstract class yang semua methodnya abstract:

Python interface_kendaraan.py 
from abc import ABC, abstractmethod

class KendaraanInterface(ABC):
    """Interface untuk semua kendaraan"""
    
    @abstractmethod
    def nyalakan_mesin(self):
        pass
    
    @abstractmethod
    def matikan_mesin(self):
        pass
    
    @abstractmethod
    def bergerak(self):
        pass
    
    @abstractmethod
    def berhenti(self):
        pass

Implementasi Interface 🚗

Sekarang kita buat berbagai jenis kendaraan yang mengimplementasi interface tersebut:

Python implementasi_kendaraan.py 
# File lengkap implementasi interface kendaraan
from abc import ABC, abstractmethod

class KendaraanInterface(ABC):
    """Interface untuk semua kendaraan"""
    
    @abstractmethod
    def nyalakan_mesin(self):
        pass
    
    @abstractmethod
    def matikan_mesin(self):
        pass
    
    @abstractmethod
    def bergerak(self):
        pass
    
    @abstractmethod
    def berhenti(self):
        pass

class Mobil(KendaraanInterface):
    def __init__(self, merk):
        self.merk = merk
        self.mesin_hidup = False
    
    def nyalakan_mesin(self):
        self.mesin_hidup = True
        return f"🚗 Mesin {self.merk} dinyalakan: Vroom!"
    
    def matikan_mesin(self):
        self.mesin_hidup = False
        return f"🚗 Mesin {self.merk} dimatikan"
    
    def bergerak(self):
        if self.mesin_hidup:
            return f"🚗 {self.merk} melaju di jalan raya"
        return f"❌ Nyalakan mesin dulu!"
    
    def berhenti(self):
        return f"🚗 {self.merk} berhenti dengan rem"

class Motor(KendaraanInterface):
    def __init__(self, merk):
        self.merk = merk
        self.mesin_hidup = False
    
    def nyalakan_mesin(self):
        self.mesin_hidup = True
        return f"🏍️ Mesin {self.merk} dinyalakan: Brum!"
    
    def matikan_mesin(self):
        self.mesin_hidup = False
        return f"🏍️ Mesin {self.merk} dimatikan"
    
    def bergerak(self):
        if self.mesin_hidup:
            return f"🏍️ {self.merk} melaju dengan lincah"
        return f"❌ Nyalakan mesin dulu!"
    
    def berhenti(self):
        return f"🏍️ {self.merk} berhenti dengan rem cakram"

# Contoh penggunaan
if __name__ == "__main__":
    print("=== Demo Interface Kendaraan ===")
    
    # Membuat objek mobil dan motor
    mobil = Mobil("Toyota")
    motor = Motor("Honda")
    
    # Demo mobil
    print("--- Demo Mobil ---")
    print(mobil.nyalakan_mesin())
    print(mobil.bergerak())
    print(mobil.berhenti())
    print(mobil.matikan_mesin())
    
    print("\n--- Demo Motor ---")
    print(motor.nyalakan_mesin())
    print(motor.bergerak())
    print(motor.berhenti())
    print(motor.matikan_mesin())

Studi Kasus: Sistem Pembayaran 💳

Mari buat sistem pembayaran yang bisa menggunakan berbagai metode pembayaran:

Python payment_system.py 
from abc import ABC, abstractmethod

# Interface untuk metode pembayaran
class PaymentInterface(ABC):
    @abstractmethod
    def proses_pembayaran(self, jumlah):
        pass
    
    @abstractmethod
    def verifikasi(self):
        pass

# Implementasi berbagai metode pembayaran
class KartuKredit(PaymentInterface):
    def __init__(self, nomor_kartu, nama_pemilik):
        self.nomor_kartu = nomor_kartu
        self.nama_pemilik = nama_pemilik
    
    def verifikasi(self):
        return f"✅ Kartu kredit {self.nomor_kartu[-4:]} terverifikasi"
    
    def proses_pembayaran(self, jumlah):
        return f"💳 Pembayaran Rp{jumlah:,} berhasil via kartu kredit"

class EWallet(PaymentInterface):
    def __init__(self, nomor_hp, saldo):
        self.nomor_hp = nomor_hp
        self.saldo = saldo
    
    def verifikasi(self):
        return f"✅ E-Wallet {self.nomor_hp} terverifikasi"
    
    def proses_pembayaran(self, jumlah):
        if self.saldo >= jumlah:
            self.saldo -= jumlah
            return f"📱 Pembayaran Rp{jumlah:,} berhasil via E-Wallet"
        return f"❌ Saldo tidak mencukupi"

class BankTransfer(PaymentInterface):
    def __init__(self, nomor_rekening, nama_bank):
        self.nomor_rekening = nomor_rekening
        self.nama_bank = nama_bank
    
    def verifikasi(self):
        return f"✅ Rekening {self.nama_bank} terverifikasi"
    
    def proses_pembayaran(self, jumlah):
        return f"🏦 Transfer Rp{jumlah:,} berhasil via {self.nama_bank}"

Processor Pembayaran 🔄

Sekarang kita buat class yang bisa memproses pembayaran dari metode apapun:

Python sistem_pembayaran_lengkap.py 
# Sistem pembayaran lengkap dengan semua class
from abc import ABC, abstractmethod

# Interface untuk metode pembayaran
class PaymentInterface(ABC):
    @abstractmethod
    def proses_pembayaran(self, jumlah):
        pass
    
    @abstractmethod
    def verifikasi(self):
        pass

# Implementasi berbagai metode pembayaran
class KartuKredit(PaymentInterface):
    def __init__(self, nomor_kartu, nama_pemilik):
        self.nomor_kartu = nomor_kartu
        self.nama_pemilik = nama_pemilik
    
    def verifikasi(self):
        return f"✅ Kartu kredit {self.nomor_kartu[-4:]} terverifikasi"
    
    def proses_pembayaran(self, jumlah):
        return f"💳 Pembayaran Rp{jumlah:,} berhasil via kartu kredit"

class EWallet(PaymentInterface):
    def __init__(self, nomor_hp, saldo):
        self.nomor_hp = nomor_hp
        self.saldo = saldo
    
    def verifikasi(self):
        return f"✅ E-Wallet {self.nomor_hp} terverifikasi"
    
    def proses_pembayaran(self, jumlah):
        if self.saldo >= jumlah:
            self.saldo -= jumlah
            return f"📱 Pembayaran Rp{jumlah:,} berhasil via E-Wallet (Sisa: Rp{self.saldo:,})"
        return f"❌ Saldo tidak mencukupi (Saldo: Rp{self.saldo:,})"

class BankTransfer(PaymentInterface):
    def __init__(self, nomor_rekening, nama_bank):
        self.nomor_rekening = nomor_rekening
        self.nama_bank = nama_bank
    
    def verifikasi(self):
        return f"✅ Rekening {self.nama_bank} terverifikasi"
    
    def proses_pembayaran(self, jumlah):
        return f"🏦 Transfer Rp{jumlah:,} berhasil via {self.nama_bank}"

class PaymentProcessor:
    def __init__(self):
        self.riwayat_transaksi = []
    
    def proses_transaksi(self, payment_method: PaymentInterface, jumlah):
        """Memproses transaksi dengan metode pembayaran apapun"""
        print("🔄 Memulai proses transaksi...")
        
        # Verifikasi metode pembayaran
        verifikasi = payment_method.verifikasi()
        print(verifikasi)
        
        # Proses pembayaran
        hasil = payment_method.proses_pembayaran(jumlah)
        print(hasil)
        
        # Tentukan status berdasarkan hasil
        status = 'Berhasil' if '❌' not in hasil else 'Gagal'
        
        # Simpan ke riwayat
        self.riwayat_transaksi.append({
            'metode': payment_method.__class__.__name__,
            'jumlah': jumlah,
            'status': status
        })
        
        print(f"Status: {status}")
        print("-" * 50)
        return hasil
    
    def tampilkan_riwayat(self):
        print("\n📊 Riwayat Transaksi:")
        for i, transaksi in enumerate(self.riwayat_transaksi, 1):
            print(f"{i}. {transaksi['metode']}: Rp{transaksi['jumlah']:,} - {transaksi['status']}")

# Contoh penggunaan lengkap
if __name__ == "__main__":
    print("=== Demo Sistem Pembayaran ===\n")
    
    # Inisialisasi processor
    processor = PaymentProcessor()
    
    # Berbagai metode pembayaran
    kartu = KartuKredit("1234-5678-9012-3456", "John Doe")
    ewallet = EWallet("081234567890", 500000)
    bank = BankTransfer("1234567890", "BCA")
    
    # Proses berbagai transaksi
    processor.proses_transaksi(kartu, 100000)
    processor.proses_transaksi(ewallet, 50000)
    processor.proses_transaksi(bank, 200000)
    processor.proses_transaksi(ewallet, 600000)  # Akan gagal karena saldo kurang
    
    # Tampilkan riwayat
    processor.tampilkan_riwayat()
