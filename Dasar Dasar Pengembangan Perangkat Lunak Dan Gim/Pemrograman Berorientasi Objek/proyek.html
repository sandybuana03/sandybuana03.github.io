🧠 Konsep Dasar Class & Object

Apa itu Class?

Class adalah template atau blueprint (cetak biru) untuk membuat object. Seperti cetakan kue yang bisa digunakan untuk membuat banyak kue dengan bentuk sama, tapi setiap kue bisa memiliki rasa dan warna yang berbeda.

Contoh: Class "Siswa" mendefinisikan bahwa setiap siswa memiliki nama, NIS, dan kelas. Dari class ini kita bisa membuat object siswa bernama Andi (NIS: 001), Budi (NIS: 002), Citra (NIS: 003), dll.

Apa itu Object?

Object adalah instance (hasil jadi) dari class. Setiap object memiliki data (atribut) dan perilaku (method) yang unik, meskipun dibuat dari class yang sama. Object adalah "benda nyata" yang bisa kita gunakan dalam program.

Contoh: Dari class "Siswa", kita bisa membuat object siswa1 (Andi, NIS: 001, nilai: 85) dan siswa2 (Budi, NIS: 002, nilai: 90). Keduanya adalah object berbeda dengan data berbeda.

📝 Syntax Dasar Class Python

Python - class_dasar.py 
# Membuat class sederhana
class Siswa:
    # Constructor - method khusus yang dijalankan saat object dibuat
    def __init__(self, nama, nis):
        # self merujuk pada object yang sedang dibuat
        self.nama = nama        # Atribut untuk menyimpan nama siswa
        self.nis = nis          # Atribut untuk menyimpan NIS siswa
        self.status = "Aktif"   # Atribut default untuk semua siswa baru
    
    # Method untuk menampilkan informasi siswa
    def tampilkan_info(self):
        return f"Nama: {self.nama}, NIS: {self.nis}, Status: {self.status}"
    
    # Method untuk mengubah status siswa
    def ubah_status(self, status_baru):
        self.status = status_baru
        return f"Status {self.nama} diubah menjadi {status_baru}"

# Membuat object (instance) dari class Siswa
siswa1 = Siswa("Andi Pratama", "001")  # Object pertama
siswa2 = Siswa("Budi Santoso", "002")  # Object kedua

# Menggunakan method dari object
print(siswa1.tampilkan_info())  # Output: Nama: Andi Pratama, NIS: 001, Status: Aktif
print(siswa2.ubah_status("Lulus"))  # Output: Status Budi Santoso diubah menjadi Lulus
print(siswa2.tampilkan_info())  # Output: Nama: Budi Santoso, NIS: 002, Status: Lulus

class Siswa:
    def __init__(self, nama, nis, kelas, tanggal_lahir="", alamat=""):
        # Atribut wajib - harus diisi saat membuat object
        self.nama = nama
        self.nis = nis
        self.kelas = kelas
        
        # Atribut opsional - boleh kosong (default value = "")
        self.tanggal_lahir = tanggal_lahir
        self.alamat = alamat
        
        # Atribut default - otomatis terisi untuk semua siswa baru
        self.nilai = {}  # Dictionary kosong untuk menyimpan nilai per mata pelajaran
        self.status = "Aktif"
        self.tahun_masuk = 2024
        
        # Pesan konfirmasi saat siswa berhasil dibuat
        print(f"✅ Siswa {self.nama} berhasil didaftarkan di kelas {self.kelas}")
    
    def tambah_nilai(self, mata_pelajaran, nilai):
        """Method untuk menambahkan nilai mata pelajaran"""
        # Validasi nilai harus antara 0-100
        if 0 <= nilai <= 100:
            self.nilai[mata_pelajaran] = nilai
            return f"✅ Nilai {mata_pelajaran} untuk {self.nama}: {nilai}"
        else:
            return "❌ Nilai harus antara 0-100!"
    
    def hitung_rata_rata(self):
        """Method untuk menghitung rata-rata semua nilai"""
        if not self.nilai:  # Jika dictionary nilai masih kosong
            return 0
        
        # Hitung total semua nilai
        total_nilai = sum(self.nilai.values())
        jumlah_mapel = len(self.nilai)
        rata_rata = total_nilai / jumlah_mapel
        
        return round(rata_rata, 2)  # Bulatkan ke 2 angka desimal
    
    def dapatkan_predikat(self):
        """Method untuk mendapatkan predikat berdasarkan rata-rata nilai"""
        rata_rata = self.hitung_rata_rata()
        
        # Sistem penilaian standar sekolah
        if rata_rata >= 90:
            return "A - Sangat Baik"
        elif rata_rata >= 80:
            return "B - Baik"
        elif rata_rata >= 70:
            return "C - Cukup"
        elif rata_rata >= 60:
            return "D - Kurang"
        else:
            return "E - Sangat Kurang"
    
    def tampilkan_info(self):
        """Method untuk menampilkan informasi lengkap siswa dalam format tabel"""
        info = f"""
╔══════════════════════════════════════╗
║           INFORMASI SISWA            ║
╠══════════════════════════════════════╣
║ Nama         : {self.nama:<20} ║
║ NIS          : {self.nis:<20} ║
║ Kelas        : {self.kelas:<20} ║
║ Status       : {self.status:<20} ║
║ Rata-rata    : {self.hitung_rata_rata():<20} ║
║ Predikat     : {self.dapatkan_predikat():<20} ║
╚══════════════════════════════════════╝
        """
        return info

🏫 Class ManajemenKelas

Python - manajemen_kelas.py 
class ManajemenKelas:
    def __init__(self, nama_kelas, wali_kelas):
        # Atribut dasar kelas
        self.nama_kelas = nama_kelas
        self.wali_kelas = wali_kelas
        self.daftar_siswa = []  # List untuk menyimpan siswa
        self.kapasitas_max = 30
        
        print(f"✅ Kelas {self.nama_kelas} berhasil dibuat!")
    
    def tambah_siswa(self, siswa):
        """Menambahkan siswa ke dalam kelas"""
        # Cek kapasitas
        if len(self.daftar_siswa) >= self.kapasitas_max:
            return "❌ Kelas sudah penuh!"
        
        # Cek NIS duplikat
        for s in self.daftar_siswa:
            if s.nis == siswa.nis:
                return "❌ NIS sudah ada!"
        
        # Tambahkan siswa
        self.daftar_siswa.append(siswa)
        siswa.kelas = self.nama_kelas
        return f"✅ {siswa.nama} berhasil ditambahkan"
    
    def hapus_siswa(self, nis):
        """Menghapus siswa berdasarkan NIS"""
        for i, siswa in enumerate(self.daftar_siswa):
            if siswa.nis == nis:
                nama = siswa.nama
                siswa.kelas = ""  # Reset kelas siswa
                self.daftar_siswa.pop(i)
                return f"✅ {nama} berhasil dihapus"
        return "❌ Siswa tidak ditemukan!"
    
    def cari_siswa(self, nis):
        """Mencari siswa berdasarkan NIS"""
        for siswa in self.daftar_siswa:
            if siswa.nis == nis:
                return siswa
        return None
    
    def hitung_rata_rata_kelas(self):
        """Menghitung rata-rata nilai kelas"""
        if not self.daftar_siswa:
            return 0
        
        total = 0
        jumlah = 0
        
        for siswa in self.daftar_siswa:
            rata_rata = siswa.hitung_rata_rata()
            if rata_rata > 0:
                total += rata_rata
                jumlah += 1
        
        return round(total / jumlah, 2) if jumlah > 0 else 0
    
    def siswa_terbaik(self):
        """Mencari siswa dengan nilai tertinggi"""
        if not self.daftar_siswa:
            return None
        
        terbaik = None
        nilai_tertinggi = 0
        
        for siswa in self.daftar_siswa:
            rata_rata = siswa.hitung_rata_rata()
            if rata_rata > nilai_tertinggi:
                nilai_tertinggi = rata_rata
                terbaik = siswa
        
        return terbaik
    
    def tampilkan_daftar_siswa(self):
        """Menampilkan daftar siswa"""
        if not self.daftar_siswa:
            return "Belum ada siswa di kelas ini."
        
        hasil = f"\n📋 DAFTAR SISWA KELAS {self.nama_kelas}\n"
        hasil += "=" * 40 + "\n"
        
        for i, siswa in enumerate(self.daftar_siswa, 1):
            rata_rata = siswa.hitung_rata_rata()
            hasil += f"{i}. {siswa.nama} (NIS: {siswa.nis}) - Rata-rata: {rata_rata}\n"
        
        return hasil
    
    def tampilkan_statistik(self):
        """Menampilkan statistik kelas"""
        jumlah_siswa = len(self.daftar_siswa)
        rata_rata_kelas = self.hitung_rata_rata_kelas()
        terbaik = self.siswa_terbaik()
        
        print(f"\n📊 STATISTIK KELAS {self.nama_kelas}")
        print("=" * 35)
        print(f"Wali Kelas      : {self.wali_kelas}")
        print(f"Jumlah Siswa    : {jumlah_siswa}/{self.kapasitas_max}")
        print(f"Rata-rata Kelas : {rata_rata_kelas}")
        print(f"Siswa Terbaik   : {terbaik.nama if terbaik else 'Belum ada'}")
        print("=" * 35)

📋 Kesimpulan Pembelajaran

🌟 Poin Penting yang Harus Diingat:

1. Class adalah Template: Seperti cetakan kue yang bisa digunakan berulang kali untuk membuat object

2. Object adalah Instance: Hasil jadi dari class dengan data unik masing-masing

3. Constructor (__init__): Method khusus yang otomatis dijalankan saat object dibuat

4. Atribut menyimpan Data: Informasi yang dimiliki setiap object (self.nama, self.nis, dll)

5. Method adalah Aksi: Fungsi yang bisa dilakukan object (tampilkan_info, tambah_nilai, dll)

6. Self merujuk Object: Keyword khusus yang merujuk pada object yang sedang digunakan

🔑 Konsep Kunci OOP:

Encapsulation: Data dan method dikemas dalam satu class

Abstraction: Menyembunyikan detail kompleks di balik interface sederhana

Reusability: Class bisa digunakan berulang untuk membuat banyak object

Modularity: Kode terorganisir dalam unit-unit yang terpisah dan jelas
